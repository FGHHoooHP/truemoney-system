async function receiveVoucher() { const input = document.getElementById("voucher"); const status = document.getElementById("status"); const voucher = input.value.trim(); status.style.display = "block"; if (!voucher) { status.textContent = "กรุณาใส่ลิงก์ซองก่อน"; return; } try { status.textContent = "กำลังส่งข้อมูล..."; const response = await fetch( "https://truemoney-system-production.up.railway.app", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ voucher: voucher }) } ); const data = await response.json(); status.textContent = data.message; } catch (error) { status.textContent = "ไม่สามารถเชื่อมต่อเซิร์ฟเวอร์ได้"; console.error(error); } }
